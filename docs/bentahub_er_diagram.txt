// --- IJAN ---
Table users { id int [pk, increment]  full_name varchar(100) [not null]  email varchar(120) [unique, not null]  password_hash varchar(255) [not null]  phone_number varchar(20)  is_verified tinyint [default: 0]  created_at timestamp }
Table roles { id int [pk, increment]  role_name varchar(50) [unique, not null]  description varchar(255) }
Table user_roles { user_id int [pk]  role_id int [pk] }

// --- IB-IB ---
Table categories { id int [pk, increment]  category_name varchar(100) [unique, not null]  description varchar(255) }
Table item_conditions { id int [pk, increment]  condition_name varchar(50) [unique, not null]  description varchar(255) }
Table safe_meetup_zones { id int [pk, increment]  zone_name varchar(100) [not null]  address varchar(255) [not null]  city varchar(100) [not null]  is_geofenced tinyint [default: 1] }

// --- DINGLASAN ---
Table listings { id int [pk, increment]  seller_id int [not null]  category_id int [not null]  condition_id int [not null]  title varchar(150) [not null]  description text  price decimal(10,2) [not null]  status enum  created_at timestamp }
Table listing_images { id int [pk, increment]  listing_id int [not null]  image_url varchar(255) [not null]  is_primary tinyint [default: 0] }
Table price_drop_alerts { id int [pk, increment]  user_id int [not null]  listing_id int [not null]  target_price decimal(10,2) [not null]  created_at timestamp }

// --- ESPIRITU ---
Table chat_threads { id int [pk, increment]  listing_id int [not null]  buyer_id int [not null]  seller_id int [not null]  created_at timestamp }
Table chat_messages { id int [pk, increment]  thread_id int [not null]  sender_id int [not null]  message_body text [not null]  sent_at timestamp }
Table meetup_schedules { id int [pk, increment]  thread_id int [not null]  zone_id int [not null]  scheduled_time datetime [not null]  status enum }

// --- DELVO ---
Table seller_trust_scores { id int [pk, increment]  seller_id int [unique, not null]  rating_score decimal(3,2)  total_reviews int  trust_tier varchar(50) }
Table reviews { id int [pk, increment]  meetup_id int [unique, not null]  reviewer_id int [not null]  seller_id int [not null]  rating tinyint [not null]  comment varchar(255)  created_at timestamp }
Table audit_logs { id int [pk, increment]  user_id int  action varchar(100) [not null]  details text  created_at timestamp }
